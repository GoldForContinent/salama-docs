create table public.locker_documents (
  id uuid not null default gen_random_uuid (),
  user_id uuid not null,
  document_name text not null,
  document_type text not null,
  file_path text not null,
  file_size integer not null,
  mime_type text not null,
  storage_url text not null,
  description text null,
  tags text[] null default '{}'::text[],
  is_archived boolean null default false,
  uploaded_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint locker_documents_pkey primary key (id),
  constraint locker_documents_user_id_fkey foreign KEY (user_id) references auth.users (id) on delete CASCADE,
  constraint valid_document_type check (
    (
      document_type = any (
        array[
          'national_id'::text,
          'passport'::text,
          'alien_id'::text,
          'refugee_id'::text,
          'military_id'::text,
          'driving_license'::text,
          'logbook'::text,
          'psi_certificate'::text,
          'towing_permit'::text,
          'badge'::text,
          'kcpe_certificate'::text,
          'kcse_certificate'::text,
          'university_degree'::text,
          'college_diploma'::text,
          'transcript'::text,
          'student_id'::text,
          'work_permit'::text,
          'professional_license'::text,
          'practicing_certificate'::text,
          'tax_pin'::text,
          'business_permit'::text,
          'title_deed'::text,
          'lease_agreement'::text,
          'allotment_letter'::text,
          'court_order'::text,
          'power_attorney'::text,
          'bank_card'::text,
          'checkbook'::text,
          'loan_agreement'::text,
          'insurance_policy'::text,
          'birth_certificate'::text,
          'death_certificate'::text,
          'marriage_certificate'::text,
          'medical_report'::text,
          'nhif_card'::text,
          'will'::text,
          'adoption_papers'::text,
          'guardianship'::text,
          'other'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_locker_documents_user_id on public.locker_documents using btree (user_id) TABLESPACE pg_default;

create index IF not exists idx_locker_documents_uploaded_at on public.locker_documents using btree (uploaded_at desc) TABLESPACE pg_default;

create index IF not exists idx_locker_documents_is_archived on public.locker_documents using btree (is_archived) TABLESPACE pg_default;

create trigger update_locker_documents_updated_at BEFORE
update on locker_documents for EACH row
execute FUNCTION update_updated_at_column ();
