create table public.reports (
  id uuid not null default extensions.uuid_generate_v4 (),
  user_id uuid not null,
  report_type character varying(5) not null,
  status character varying(20) not null default 'active'::character varying,
  matched_report_id uuid null,
  full_name text not null,
  phone text not null,
  email text not null,
  timeline character varying(10) null,
  location_description text not null,
  additional_details text null,
  collection_point text null,
  reward_amount integer null,
  recovery_fee integer null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  delivery_status character varying(20) null default 'unclaimed_unverified'::character varying,
  constraint reports_pkey primary key (id),
  constraint reports_matched_report_id_fkey foreign KEY (matched_report_id) references reports (id),
  constraint reports_user_id_fkey foreign KEY (user_id) references auth.users (id),
  constraint reports_phone_check check ((phone ~ '^(07|01)\d{8}$'::text)),
  constraint reports_email_check check ((email ~ '^[^@]+@[^@]+\.[^@]+$'::text)),
  constraint reports_report_type_check check (
    (
      (report_type)::text = any (
        (
          array[
            'lost'::character varying,
            'found'::character varying
          ]
        )::text[]
      )
    )
  ),
  constraint reports_status_check check (
    (
      (status)::text = any (
        (
          array[
            'active'::character varying,
            'potential_match'::character varying,
            'matched_successfully'::character varying,
            'payment_pending'::character varying,
            'completed'::character varying
          ]
        )::text[]
      )
    )
  ),
  constraint reports_timeline_check check (
    (
      (timeline)::text = any (
        (
          array[
            'today'::character varying,
            'yesterday'::character varying,
            'week'::character varying,
            'month'::character varying
          ]
        )::text[]
      )
    )
  ),
  constraint reports_delivery_status_check check (
    (
      (delivery_status)::text = any (
        (
          array[
            'unclaimed_unverified'::character varying,
            'unclaimed_verified'::character varying,
            'claimed'::character varying
          ]
        )::text[]
      )
    )
  )
) TABLESPACE pg_default;
